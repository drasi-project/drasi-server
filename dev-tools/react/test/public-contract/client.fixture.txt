export const freshHeaders: DrasiClientOptions = {
  ...options,
  async headers(context) {
    const signal: AbortSignal = context.signal;
    const transport: 'rest' | 'sse' = context.transport;
    const url: string = context.url;
    // @ts-expect-error Request context does not expose arbitrary transport fields.
    context.defaultInstance;
    signal.throwIfAborted();
    return { 'X-Transport': transport, 'X-Endpoint': url };
  },
  eventSourceFactory(url, transport) {
    const endpoint: string = url;
    const headers: Headers = transport.headers;
    const credentials: RequestCredentials = transport.credentials;
    const signal: AbortSignal = transport.signal;
    // @ts-expect-error SSE credentials are typed, not an unstructured value.
    const secret: number = transport.credentials;
    void [endpoint, headers, credentials, signal, secret];
    return { onopen: null, onmessage: null, onerror: null, addEventListener() {}, close() {} };
  },
};

export async function typedClientCallbacks(client: DrasiClient) {
  const untypedRows: ResultRow[] = await client.getQueryResults('temperatures');
  // @ts-expect-error Default REST values cannot be consumed as known numbers.
  const unsafeNumber: number = untypedRows[0].celsius;
  // @ts-expect-error A transport read cannot establish an application's row schema.
  const temperatures: Temperature[] = await client.getQueryResults('temperatures');
  const unsubscribe = client.subscribe('temperatures', result => {
    const wire: QueryResult = result;
    const raw: ResultRow[] = result.data;
    // @ts-expect-error Default subscription values are unknown, not any.
    result.data[0].celsius.toFixed(1);
    void [wire, raw];
  }, error => {
    const errorCode: DrasiErrorCode = error.code;
    // @ts-expect-error Typed error callbacks do not expose arbitrary properties.
    error.serverSecret;
    void errorCode;
  });
  void [unsafeNumber, temperatures];
  return unsubscribe;
}

export function typedBatches(temperatures: QueryResult<Temperature>, deliveries: QueryResult<Delivery>) {
  const celsius: number = temperatures.data[0].celsius;
  const parcels: number = deliveries.data[0].parcels;
  // @ts-expect-error Generic batches keep distinct non-Trading row models separate.
  const crossed: QueryResult<Delivery> = temperatures;
  // @ts-expect-error Generic rows do not expose the other domain's properties.
  temperatures.data[0].routeId;
  void [celsius, parcels, crossed];
}
