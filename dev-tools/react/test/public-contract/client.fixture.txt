export async function typedClientCallbacks(client: DrasiClient) {
  const untypedRows: ResultRow[] = await client.getQueryResults('temperatures');
  // @ts-expect-error Default REST values cannot be consumed as known numbers.
  const unsafeNumber: number = untypedRows[0].celsius;
  // @ts-expect-error A transport read cannot establish an application's row schema.
  const temperatures: Temperature[] = await client.getQueryResults('temperatures');
  const unsubscribe: QuerySubscription = client.subscribe('temperatures', result => {
    const wire: QueryResult = result;
    if (result.kind === 'snapshot') {
      const raw: readonly ResultRow[] = result.rows;
      // @ts-expect-error Default subscription values are unknown, not any.
      result.rows[0].celsius.toFixed(1);
      // @ts-expect-error A snapshot cannot be consumed as a delta.
      result.changes;
      void raw;
    } else {
      const changes: readonly ResultChange[] = result.changes;
      const sourceTime: number | undefined = result.sourceTimestamp;
      // @ts-expect-error A delta does not contain a complete row set.
      result.rows;
      void [changes, sourceTime];
    }
    // @ts-expect-error Legacy data/snapshot/timestamp batches are no longer the public result contract.
    result.data;
    void wire;
  }, error => {
    const errorCode: DrasiErrorCode = error.code;
    // @ts-expect-error Typed error callbacks do not expose arbitrary properties.
    error.serverSecret;
    void errorCode;
  }, next => {
    const state: QuerySubscriptionState = next;
    const scope: QueryErrorScope | null = state.errorScope;
    const error: DrasiError | null = state.error;
    const stale: boolean = state.stale;
    // @ts-expect-error Subscription state cannot infer the hook's projected empty state.
    const empty: 'empty' = state.status;
    void [scope, error, stale, empty];
  });
  const cleanup: () => void = unsubscribe;
  const retry: () => void = unsubscribe.retry;
  const current: QuerySubscriptionState = unsubscribe.getState();
  void [unsafeNumber, temperatures, cleanup, retry, current];
  return unsubscribe;
}

export function typedBatches(temperatures: QueryResult<Temperature>, deliveries: QueryResult<Delivery>) {
  if (temperatures.kind === 'snapshot' && deliveries.kind === 'snapshot') {
    const celsius: number = temperatures.rows[0].celsius;
    const parcels: number = deliveries.rows[0].parcels;
    // @ts-expect-error Generic rows do not expose the other domain's properties.
    temperatures.rows[0].routeId;
    void [celsius, parcels];
  }
  // @ts-expect-error Generic batches keep distinct non-Trading row models separate.
  const crossed: QueryResult<Delivery> = temperatures;
  void crossed;
}

export function typedChanges(change: ResultChange<Temperature>): number {
  switch (change.kind) {
    case 'upsert':
      // @ts-expect-error An upsert has no before row.
      change.before;
      return change.after.celsius;
    case 'update': {
      const before: Temperature = change.before;
      const after: Temperature = change.after;
      // @ts-expect-error Both sides of an update retain the chosen domain model.
      const delivery: Delivery = change.before;
      void delivery;
      return after.celsius - before.celsius;
    }
    case 'delete':
      // @ts-expect-error A delete has no after row.
      change.after;
      return change.before.celsius;
    default: {
      const exhaustive: never = change;
      return exhaustive;
    }
  }
}

export function typedStreamCallbacks(stream: DrasiSSEClient): () => void {
  const error: DrasiError | null = stream.getQueryError('temperatures');
  const cleanup = stream.subscribe('temperatures', result => {
    const delta: QueryDelta = result;
    // @ts-expect-error The low-level stream does not manufacture REST snapshots.
    const snapshot: QuerySnapshot = result;
    void [delta, snapshot];
  }, failure => {
    const code: DrasiErrorCode = failure.code;
    void code;
  });
  // @ts-expect-error Query-local REST retry belongs to DrasiClient subscriptions, not raw SSE cleanup.
  cleanup.retry();
  void error;
  return cleanup;
}
