interface TemperatureWire {
  stationId: string;
  millicelsius: number;
}

interface Temperature {
  stationId: string;
  celsius: number;
}

interface Delivery {
  routeId: string;
  parcels: number;
}

const clientOptions: DrasiClientOptions = {
  serverUrl: 'https://drasi.example', instanceId: 'warehouse-west', queryIds: ['temperatures'],
  reaction: { id: 'warehouse-events', endpoint: 'https://events.example/warehouse' },
  credentials: 'same-origin', headers: { 'X-Consumer': 'packed-hooks' },
  resultAdapter: sse034ResultAdapter, reconciliation: { maxPendingChanges: 10000 },
};

export const temperatureOptions: UseDrasiQueryOptions<Temperature> = {
  transform(row) {
    // @ts-expect-error Transform input remains unknown until application validation.
    const unchecked: TemperatureWire = row;
    void unchecked;
    if (typeof row.stationId !== 'string' || typeof row.millicelsius !== 'number') {
      throw new TypeError('Expected a temperature reading');
    }
    const wire: TemperatureWire = { stationId: row.stationId, millicelsius: row.millicelsius };
    return { stationId: wire.stationId, celsius: wire.millicelsius / 1000 };
  },
  getKey(row) {
    const input: Readonly<ResultRow> = row;
    // @ts-expect-error Raw keys cannot assume the transformed celsius field is numeric.
    const degrees: number = row.celsius;
    // @ts-expect-error Raw identity fields also require validation before use.
    const unchecked: number = row.millicelsius;
    if (typeof input.stationId !== 'string' || !input.stationId) throw new Error('Missing station identity');
    void [degrees, unchecked];
    return input.stationId;
  },
  postProcess(rows) {
    const output: Temperature[] = rows;
    // @ts-expect-error Post-processing uses the transformed model, not another domain's rows.
    const deliveries: Delivery[] = rows;
    void deliveries;
    return output.filter(row => row.celsius > 0);
  },
};

export const defaultInput: UseDrasiQueryOptions<Delivery> = {
  transform(raw) {
    const input: ResultRow = raw;
    // @ts-expect-error Unspecified wire input properties require narrowing.
    raw.routeId.toUpperCase();
    if (typeof input.routeId !== 'string' || typeof input.parcels !== 'number') {
      throw new TypeError('Expected a delivery');
    }
    return { routeId: input.routeId, parcels: input.parcels };
  },
  getKey: row => {
    if (typeof row.routeId !== 'string' || !row.routeId) throw new Error('Missing route identity');
    return row.routeId;
  },
  postProcess: rows => rows.filter(row => row.parcels > 0),
};

export const defaultOptions: UseDrasiQueryOptions = {
  getKey(row) {
    const input: Record<string, unknown> = row;
    // @ts-expect-error Default row values are unknown, never implicit any.
    const numeric: number = row.parcels;
    void numeric;
    if (typeof input.id !== 'string' || !input.id) throw new Error('Missing identity');
    return input.id;
  },
  transform: row => row,
};

export function Consumer(): ReactNode {
  const temperatures: UseDrasiQueryResult<Temperature> =
    useDrasiQuery<Temperature>('temperatures', temperatureOptions);
  const deliveries: UseDrasiQueryResult<Delivery> = useDrasiQuery<Delivery>('deliveries', defaultInput);
  const defaultResult: UseDrasiQueryResult = useDrasiQuery('other', defaultOptions);
  const filtered: UseDrasiQueryResult<Temperature> = useDrasiQuery('temperatures', nullableOptions);
  // @ts-expect-error Every hook call requires options, even a raw read.
  useDrasiQuery('other');
  // @ts-expect-error A typed generic does not waive the required raw key and validating transform.
  useDrasiQuery<Temperature>('temperatures');
  const rows: ResultRow[] | null = defaultResult.data;
  // @ts-expect-error Default hook rows are not a typed domain model.
  const unknownRows: Temperature[] | null = defaultResult.data;
  // @ts-expect-error Distinct non-Trading result models must remain distinct.
  const wrongRows: Delivery[] | null = temperatures.data;
  const resultError: DrasiError | null = temperatures.error;
  const lastUpdate: Date | null = deliveries.lastUpdate;
  const queryStatus: QueryStatus = temperatures.status;
  const stale: boolean = temperatures.stale;
  const scope: QueryErrorScope | null = temperatures.errorScope;
  const retryQuery: () => void = temperatures.retry;
  const definition: UseDrasiQueryDefinitionResult = useDrasiQueryDefinition('temperatures');
  const config: QueryConfig | null = definition.config;
  const definitionError: DrasiError | null = definition.error;
  const loading: boolean = definition.loading;
  const status: ConnectionStatus = useDrasiConnectionStatus();
  const url: string | null = useDrasiServerUiUrl();
  const owner: DrasiContextValue = useDrasiClient();
  const retryConnection: () => void = owner.retry;
  void [rows, unknownRows, wrongRows, resultError, lastUpdate, config, definitionError, loading, status, url, owner,
    filtered, queryStatus, stale, scope, retryQuery, retryConnection];
  return createElement('p', null, temperatures.data?.[0]?.celsius ?? 'waiting');
}
