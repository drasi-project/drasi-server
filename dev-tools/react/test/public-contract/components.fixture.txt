export const table: QueryTableProps<Temperature> = {
  queryId: 'temperatures', columns, actions, defaultSort: sort,
  rowKey: row => row.stationId,
  queryOptions: {
    getKey: raw => {
      if (typeof raw.station_id !== 'string' || !raw.station_id) throw new Error('Missing station identity');
      return raw.station_id;
    },
    transform: raw => {
      if (typeof raw.station_id !== 'string' || typeof raw.millicelsius !== 'number') {
        throw new Error('Expected a temperature');
      }
      return { stationId: raw.station_id, celsius: raw.millicelsius / 1000 };
    },
  },
  animateOnChange: 'celsius',
  rowClassName(row, index) {
    const degrees: number = row.celsius;
    const position: number = index;
    // @ts-expect-error Table callbacks do not lose their row generic.
    row.parcels;
    return degrees > 30 && position === 0 ? 'first-hot' : '';
  },
  onSortChange(next) {
    const value: SortConfig | null = next;
    // @ts-expect-error A sort request can explicitly clear sorting.
    const required: SortConfig = next;
    // @ts-expect-error Sort callbacks retain their direction union.
    const impossible: 'down' = next?.direction;
    void [value, required, impossible];
  },
  renderRow(row, definitions, animation, defaultRender) {
    const station: string = row.stationId;
    const ownColumns: readonly ColumnDef<Temperature>[] = definitions;
    const rendered: ReactNode = defaultRender();
    void [station, ownColumns, animation];
    return rendered;
  },
  renderLoading(context) {
    const ownContext: QueryTableRenderContext<Temperature> = context;
    const query: UseDrasiQueryResult<Temperature> = context.query;
    const rows: readonly Temperature[] | null = context.rows;
    const retry: () => void = context.retryConnection;
    // @ts-expect-error The full query result keeps the table's row model.
    const other: UseDrasiQueryResult<Delivery> = context.query;
    void [ownContext, query, rows, retry, other];
    return null;
  },
  renderEmpty({ query, rows }) {
    const status: QueryStatus = query.status;
    return createElement('span', null, `${status}: ${rows?.length ?? 0} readings`);
  },
  renderError(context) {
    const ownContext: QueryTableErrorContext<Temperature> = context;
    const error: DrasiError = context.error;
    const code: DrasiErrorCode = error.code;
    const scope: QueryErrorScope | null = context.query.errorScope;
    const lastUpdate: Date | null = context.query.lastUpdate;
    const queryRetry: () => void = context.query.retry;
    const connectionRetry: () => void = context.retryConnection;
    // @ts-expect-error Query error slots preserve DrasiError, not an arbitrary application error subtype.
    const applicationError: SensorError = context.error;
    void [ownContext, code, scope, lastUpdate, applicationError];
    return createElement('button', {
      onClick: scope === 'connection' ? connectionRetry : queryRetry,
    }, error.message);
  },
  renderStale(context) {
    const stale: boolean = context.query.stale;
    const retry: (() => void) | undefined = context.state.retry;
    void [stale, retry];
    return context.state.staleMessage;
  },
  renderHeader(context) {
    const header: DataTableHeaderContext<Temperature, DrasiError> = context;
    // @ts-expect-error QueryTable's header remains a presentation slot, not an enriched query-state slot.
    context.query;
    return header.defaultRender();
  },
};
